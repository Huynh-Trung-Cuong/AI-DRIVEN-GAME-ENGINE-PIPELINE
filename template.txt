from ursina import *
from dsa import StateGraph, check_minkowski_collision, check_minkowski_ccd, FrameActionHeap

# ==========================================
# KHỞI TẠO HÀNG ĐỢI ƯU TIÊN SỰ KIỆN (GLOBAL HEAP)
# ==========================================
action_queue = FrameActionHeap()

class Player1(Entity):
    def __init__(self, start_pos, controls, **kwargs):
        super().__init__(model='cube', collider='box', color=color.cyan, position=start_pos, **kwargs)
        
        # --- 1. CHỈ SỐ SINH TỒN & BIẾN HỆ THỐNG ---
        self.hp = 100
        self.controls = controls
        self.enemy_target = None
        self.description = "MÔ TẢ KỸ NĂNG CỦA PLAYER 1 (AI sẽ điền vào đây)"
        
        # --- 2. QUẢN LÝ COOLDOWN ---
        self.cooldown1 = 0.0
        self.cooldown2 = 0.0
        
        # --- 3. KHỞI TẠO STATE GRAPH (FSM) ---
        self.sm = StateGraph("IDLE")
        self.setup_state_machine()
        
        # --- 4. CÁC BIẾN CƠ CHẾ KHÁC (AI tự thêm nếu cần) ---
        self.move_speed = 5
        self.old_pos = self.position

    def setup_state_machine(self):
        """Thiết lập đồ thị trạng thái và các cạnh có điều kiện"""
        states = ["IDLE", "MOVE", "ATTACK", "SKILL1", "SKILL2", "HITSTUN"]
        for s in states:
            self.sm.add_state(s)
            
        # Điều kiện chuyển trạng thái (Guarded Edges)
        self.sm.add_edge("IDLE", "MOVE")
        self.sm.add_edge("MOVE", "IDLE")
        
        # Chỉ đánh/dùng chiêu được khi đang ở IDLE hoặc MOVE và hết CD
        for base_state in ["IDLE", "MOVE"]:
            self.sm.add_edge(base_state, "ATTACK")
            self.sm.add_edge(base_state, "SKILL1", lambda: self.cooldown1 <= 0)
            self.sm.add_edge(base_state, "SKILL2", lambda: self.cooldown2 <= 0)
        
        # Mọi trạng thái đều có thể bị ngắt bởi HITSTUN
        for s in states:
            self.sm.add_edge(s, "HITSTUN")
            
        # Hết HITSTUN tự về IDLE
        self.sm.add_edge("HITSTUN", "IDLE")

    def update(self):
        # 1. Lưu vị trí cũ cho thuật toán CCD (Chống xuyên thấu)
        self.old_pos = self.position
        
        # 2. Cập nhật Cooldown
        if self.cooldown1 > 0: self.cooldown1 -= time.dt
        if self.cooldown2 > 0: self.cooldown2 -= time.dt
            
        # 3. Logic Di chuyển
        if self.sm.current_state in ["IDLE", "MOVE"]:
            move_dir = Vec3(0, 0, 0)
            if held_keys[self.controls['up']]: move_dir.z += 1
            if held_keys[self.controls['down']]: move_dir.z -= 1
            if held_keys[self.controls['right']]: move_dir.x += 1
            if held_keys[self.controls['left']]: move_dir.x -= 1
            
            if move_dir != Vec3(0, 0, 0):
                self.position += move_dir.normalized() * self.move_speed * time.dt
                self.sm.request_transition("MOVE")
            else:
                self.sm.request_transition("IDLE")

        # 4. Bắt Input Kỹ Năng
        if held_keys[self.controls['attack']]: self.execute_attack()
        if held_keys[self.controls['skill1']]: self.execute_skill1()
        if held_keys[self.controls['skill2']]: self.execute_skill2()

    # ==========================================
    # LOGIC KỸ NĂNG: ĐÁNH THƯỜNG
    # ==========================================
    def execute_attack(self):
        if self.sm.request_transition("ATTACK"):
            invoke(self.attack_active, delay=0.1)

    def attack_active(self):
        if self.sm.current_state != "ATTACK": return 
        
        # Check va chạm tĩnh Minkowski
        is_hit = check_minkowski_collision(self.position, self.scale, self.enemy_target.position, self.enemy_target.scale)
        
        if is_hit:
            # SỬ DỤNG HEAP: Push hành động vào hàng đợi thay vì gọi trực tiếp
            # Priority 1 (Cao nhất): Áp dụng hiệu ứng CC/Hitstun trước
            action_queue.push_action(1, self.enemy_target.apply_hitstun, 0.2)
            
            # Priority 2: Trừ máu
            def take_damage(target, amount): target.hp -= amount
            action_queue.push_action(2, take_damage, self.enemy_target, 5)

        invoke(self.reset_to_idle, delay=0.2) 

    # ==========================================
    # LOGIC KỸ NĂNG 1 (Lướt/Dash mẫu sử dụng CCD)
    # ==========================================
    def execute_skill1(self):
        if self.sm.request_transition("SKILL1"):
            self.cooldown1 = 5.0 
            # Giả lập logic lướt (Dash)
            self.position += self.forward * 5
            
            # Sử dụng CCD (Swept AABB) để check xuyên thấu khi lướt
            is_hit = check_minkowski_ccd(
                self.old_pos, self.position, self.scale, 
                self.enemy_target.old_pos, self.enemy_target.position, self.enemy_target.scale
            )
            
            if is_hit:
                action_queue.push_action(1, self.enemy_target.apply_hitstun, 0.5)
                def take_damage(target, amount): target.hp -= amount
                action_queue.push_action(2, take_damage, self.enemy_target, 15)
                
            invoke(self.reset_to_idle, delay=1.0)

    # ==========================================
    # LOGIC KỸ NĂNG 2
    # ==========================================
    def execute_skill2(self):
        if self.sm.request_transition("SKILL2"):
            self.cooldown2 = 8.0 
            invoke(self.reset_to_idle, delay=1.0)

    # ==========================================
    # HÀM TIỆN ÍCH
    # ==========================================
    def reset_to_idle(self):
        if self.sm.current_state != "HITSTUN":
            self.sm.current_state = "IDLE"

    def apply_hitstun(self, duration):
        if self.sm.request_transition("HITSTUN"):
            self.color = color.red 
            invoke(self.end_hitstun, delay=duration)
            
    def end_hitstun(self):
        self.color = color.cyan
        self.reset_to_idle()

# PLAYER 2 (Cấu trúc tương tự Player 1)
# =====================================================================
class Player2(Entity):
    def __init__(self, start_pos, controls, **kwargs):
        super().__init__(model='cube', collider='box', color=color.magenta, position=start_pos, **kwargs)
        self.hp = 100
        self.controls = controls
        self.enemy_target = None
        self.description = "MÔ TẢ KỸ NĂNG CỦA PLAYER 2"
        self.cooldown1 = 0.0
        self.cooldown2 = 0.0
        
        self.sm = StateGraph("IDLE")
        self.setup_state_machine()
        
        self.move_speed = 5
        self.old_pos = self.position

    def setup_state_machine(self):
        states = ["IDLE", "MOVE", "ATTACK", "SKILL1", "SKILL2", "HITSTUN"]
        for s in states: self.sm.add_state(s)
        self.sm.add_edge("IDLE", "MOVE")
        self.sm.add_edge("MOVE", "IDLE")
        for base_state in ["IDLE", "MOVE"]:
            self.sm.add_edge(base_state, "ATTACK")
            self.sm.add_edge(base_state, "SKILL1", lambda: self.cooldown1 <= 0)
            self.sm.add_edge(base_state, "SKILL2", lambda: self.cooldown2 <= 0)
        for s in states: self.sm.add_edge(s, "HITSTUN")
        self.sm.add_edge("HITSTUN", "IDLE")

    def update(self):
        self.old_pos = self.position
        if self.cooldown1 > 0: self.cooldown1 -= time.dt
        if self.cooldown2 > 0: self.cooldown2 -= time.dt
            
        if self.sm.current_state in ["IDLE", "MOVE"]:
            move_dir = Vec3(0, 0, 0)
            if held_keys[self.controls['up']]: move_dir.z += 1
            if held_keys[self.controls['down']]: move_dir.z -= 1
            if held_keys[self.controls['right']]: move_dir.x += 1
            if held_keys[self.controls['left']]: move_dir.x -= 1
            if move_dir != Vec3(0, 0, 0):
                self.position += move_dir.normalized() * self.move_speed * time.dt
                self.sm.request_transition("MOVE")
            else:
                self.sm.request_transition("IDLE")

        if held_keys[self.controls['attack']]: self.execute_attack()
        if held_keys[self.controls['skill1']]: self.execute_skill1()
        if held_keys[self.controls['skill2']]: self.execute_skill2()

    def execute_attack(self):
        if self.sm.request_transition("ATTACK"):
            invoke(self.reset_to_idle, delay=0.2)

    def execute_skill1(self):
        if self.sm.request_transition("SKILL1"):
            self.cooldown1 = 5.0
            invoke(self.reset_to_idle, delay=1.0)

    def execute_skill2(self):
        if self.sm.request_transition("SKILL2"):
            self.cooldown2 = 8.0
            invoke(self.reset_to_idle, delay=1.0)

    def reset_to_idle(self):
        if self.sm.current_state != "HITSTUN":
            self.sm.current_state = "IDLE"

    def apply_hitstun(self, duration):
        if self.sm.request_transition("HITSTUN"):
            self.color = color.red
            invoke(self.end_hitstun, delay=duration)
            
    def end_hitstun(self):
        self.color = color.magenta
        self.reset_to_idle()

# BỘ QUẢN LÝ HEAP (TỰ ĐỘNG XỬ LÝ SỰ KIỆN CUỐI FRAME)

class HeapProcessor(Entity):
    """Entity ẩn này sẽ chạy sau tất cả các Player để xử lý các event trong Heap"""
    def update(self):
        action_queue.process_all_actions()

heap_processor = HeapProcessor()
